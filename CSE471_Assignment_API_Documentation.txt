API 1.1: Upload Resume
────────────────────────────────────────────────────────────────
Endpoint URL: http://localhost:1008/api/resumes/upload
HTTP Method: POST
Headers:
  - Content-Type: multipart/form-data

Body/Parameters:
  - File upload (resume): PDF/DOC/DOCX file
  - userId (string, required): Unique user identifier
  - fullName (string, optional): User's full name
  - email (string, optional): User's email address
  - phone (string, optional): User's phone number

Code Snippet (Node.js - Controller):
──────────────────────────────────────
exports.uploadResume = async (req, res) => {
  try {
    if (!req.file) {
      return res.status(400).json({
        success: false,
        message: 'No file uploaded',
      });
    }

    const { userId, fullName, email, phone } = req.body;

    const fileType = req.file.mimetype.includes('pdf')
      ? 'pdf'
      : req.file.mimetype.includes('word')
      ? 'docx'
      : 'doc';

    const resume = new Resume({
      userId,
      fileName: req.file.originalname,
      fileType,
      filePath: req.file.path,
      fullName: fullName || 'Not Provided',
      email: email || 'Not Provided',
      phone: phone || 'Not Provided',
      skills: [],
      experience: [],
      education: [],
    });

    await resume.save();

    res.status(201).json({
      success: true,
      message: 'Resume uploaded successfully',
      data: {
        resumeId: resume._id,
        fileName: resume.fileName,
        uploadedAt: resume.uploadedAt,
      },
    });
  } catch (error) {
    res.status(500).json({
      success: false,
      message: 'Error uploading resume',
      error: error.message,
    });
  }
};

Code Snippet (Node.js - Controller):
──────────────────────────────────────
exports.parseResume = async (req, res) => {
  try {
    const { resumeId } = req.params;
    const { skills, experience, education, summary } = req.body;

    const resume = await Resume.findByIdAndUpdate(
      resumeId,
      {
        skills: skills || [],
        experience: experience || [],
        education: education || [],
        summary: summary || '',
      },
      { new: true }
    );

    if (!resume) {
      return res.status(404).json({
        success: false,
        message: 'Resume not found',
      });
    }

    res.status(200).json({
      success: true,
      message: 'Resume parsed successfully',
      data: {
        resumeId: resume._id,
        skills: resume.skills,
        experience: resume.experience,
        education: resume.education,
      },
    });
  } catch (error) {
    res.status(500).json({
      success: false,
      message: 'Error parsing resume',
      error: error.message,
    });
  }
};

exports.createJobPosting = async (req, res) => {
  try {
    const {
      employerId,
      jobTitle,
      description,
      requiredSkills,
      qualifications,
      location,
      salary,
      jobType,
      experience,
      deadline,
    } = req.body;

    if (!employerId || !jobTitle || !description || !location) {
      return res.status(400).json({
        success: false,
        message: 'Required fields are missing',
      });
    }

    const jobPosting = new JobPosting({
      employerId,
      jobTitle,
      description,
      requiredSkills: requiredSkills || [],
      qualifications: qualifications || [],
      location,
      salary: salary || {},
      jobType: jobType || 'Full-Time',
      experience: experience || {},
      deadline: deadline || null,
      status: 'Open',
      applicantsCount: 0,
    });

    await jobPosting.save();

    res.status(201).json({
      success: true,
      message: 'Job posting created successfully',
      data: {
        jobId: jobPosting._id,
        jobTitle: jobPosting.jobTitle,
        status: jobPosting.status,
        createdAt: jobPosting.createdAt,
      },
    });
  } catch (error) {
    res.status(500).json({
      success: false,
      message: 'Error creating job posting',
      error: error.message,
    });
  }
};

API 2.3: Get All Job Postings
────────────────────────────────────────────────────────────────
Endpoint URL: http://localhost:1008/api/jobs?status=Open&location=&jobType=Full-Time
HTTP Method: GET
Headers: None
Query Parameters (optional):
  - status: Filter by job status (Open, Closed, On Hold)
  - location: Filter by location
  - jobType: Filter by job type (Full-Time, Part-Time, Contract, Internship)

Code Snippet (Node.js - Controller):
──────────────────────────────────────
exports.getAllJobPostings = async (req, res) => {
  try {
    const { status, location, jobType } = req.query;
    let filter = {};

    if (status) filter.status = status;
    if (location) filter.location = { $regex: location, $options: 'i' };
    if (jobType) filter.jobType = jobType;

    const jobs = await JobPosting.find(filter).sort({ createdAt: -1 });

    res.status(200).json({
      success: true,
      count: jobs.length,
      data: jobs,
    });
  } catch (error) {
    res.status(500).json({
      success: false,
      message: 'Error fetching job postings',
      error: error.message,
    });
  }
};

exports.searchJobsBySkills = async (req, res) => {
  try {
    const { skills } = req.body;

    if (!skills || !Array.isArray(skills) || skills.length === 0) {
      return res.status(400).json({
        success: false,
        message: 'Skills array is required',
      });
    }

    const jobs = await JobPosting.find({
      status: 'Open',
      requiredSkills: { $in: skills },
    }).sort({ createdAt: -1 });

    res.status(200).json({
      success: true,
      count: jobs.length,

    });

SETUP & RUNNING INSTRUCTIONS
════════════════════════════════════════════════════════════════

Prerequisites:
- Node.js installed (v14 or higher)
- MongoDB running locally or accessible via connection string
- Postman (for API testing)

Installation Steps:

1. Navigate to backend directory:
   cd backend

2. Install dependencies:
   npm install

3. Create .env file (already created with default settings):
   MONGODB_URI=mongodb://localhost:27017/resume-job-portal
   PORT=1008

4. Ensure MongoDB is running:
   - For local MongoDB: mongod
   - Or update MONGODB_URI in .env for remote connection

5. Start the server:
   npm start

   Output should show:
   ========================================
   Server started successfully!
   Port: 1008
   URL: http://localhost:1008
   ========================================

6. Import Postman Collection:
   - Open Postman
   - Click "Import"
   - Select "Resume_Job_Portal_API.postman_collection.json"
   - Set variables {{resumeId}} and {{jobId}} before testing
